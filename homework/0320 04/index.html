if — 判斷成績是否及格

if 條件判斷

使用 if 判斷分數是否 ≥ 60，輸出及格或不及格。

let score = 75;

if (score >= 60) {
    console.log("及格");
} else {
    console.log("不及格");
}

if else if — 判斷成績等第

if else if 多條件判斷

依照分數範圍輸出 A / B / C / D / F 等第。

let score = 85;

if (score >= 90) {
    console.log("A");
} else if (score >= 80) {
    console.log("B");
} else if (score >= 70) {
    console.log("C");
} else if (score >= 60) {
    console.log("D");
} else {
    console.log("F");
}

for — 印出 1 到 10

for 迴圈

for 迴圈從 1 跑到 10，依序輸出數字。

for (let i = 1; i <= 10; i++) {
    console.log(i);
}

for — 計算 1 到 100 的總和

for 累加

用 for 迴圈累加 1~100，最後輸出總和。

let sum = 0;

for (let i = 1; i <= 100; i++) {
    sum = sum + i;
}

console.log("總和是：" + sum);

while — 印出 1 到 5

while 迴圈

用 while 在條件成立時重複執行程式碼。

let i = 1;
while (i <= 5) {
    console.log(i);
    i++;
}

function 函式

for + array — 遍歷水果陣列

for array

用 for 迴圈逐一讀取陣列中的水果名稱。

let fruits = ["蘋果", "香蕉", "芭樂", "西瓜"];

JSON + object + array 綜合練習

JSON JSON.parse for if

用 JSON.parse 解析 JSON 字串，再用 for + if 判斷每位學生是否及格。

let jsonText = `
[
    {"name":"小明","score":85},
    {"name":"小華","score":58},
    {"name":"小美","score":92}
]
`;

let students = JSON.parse(jsonText);

for (let i = 0; i < students.length; i++) {
    if (students[i].score >= 60) {
        console.log(students[i].name + " 及格");
    } else {
        console.log(students[i].name + " 不及格");
    }
}
