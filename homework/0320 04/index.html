if 01test.js

if — 判斷成績是否及格

if 條件判斷

使用 if 判斷分數是否 ≥ 60，輸出及格或不及格。

let score = 75;
        


if (score >= 60) {

    console.log("及格");
    
} else {

    console.log("不及格");
    
}
if else 02test.js

if else if — 判斷成績等第

if else if 多條件判斷

依照分數範圍輸出 A / B / C / D / F 等第。

let score = 85;
        


if (score >= 90) {

    console.log("A");
    
} else if (score >= 80) {

    console.log("B");
    
} else if (score >= 70) {

    console.log("C");
    
} else if (score >= 60) {

    console.log("D");
    
} else {

    console.log("F");
    
}
for 03test.js

for — 印出 1 到 10

for 迴圈

for 迴圈從 1 跑到 10，依序輸出數字。

for (let i = 1; i <= 10; i++) {
        
    console.log(i);
    
}
累加 04test.js

for — 計算 1 到 100 的總和

for 累加

用 for 迴圈累加 1~100，最後輸出總和。

let sum = 0;
        


for (let i = 1; i <= 100; i++) {

    sum = sum + i;
    
}



console.log("總和是：" + sum);
while 05test.js

while — 印出 1 到 5

while 迴圈

用 while 在條件成立時重複執行程式碼。

let i = 1;
        
while (i <= 5) {

    console.log(i);
    
    i++;
    
}
function 06test.js

function — 加法函式

function 函式

建立一個可重複使用的 add 函式，計算兩數相加。

function add(a, b) {
        
    return a + b;
    
}



let result = add(10, 20);

console.log("答案是：" + result);
array 07test.js

array — 儲存多筆成績

array 索引

用陣列儲存多個成績，並透過索引取值。

let scores = [80, 90, 70, 60, 100];
        


console.log("第一個成績：" + scores[0]);

console.log("第二個成績：" + scores[1]);

console.log("陣列長度：" + scores.length);
for+array 08test.js

for + array — 遍歷水果陣列

for array

用 for 迴圈逐一讀取陣列中的水果名稱。

let fruits = ["蘋果", "香蕉", "芭樂", "西瓜"];
        


for (let i = 0; i < fruits.length; i++) {

    console.log("水果：" + fruits[i]);
    
}
object 09test.js

object — 學生物件

object 屬性

用物件來表示一位學生的完整資料（姓名、年齡、科系）。

let student = {
        
    name: "小美",
    
    age: 20,
    
    department: "資工系"
    
};



console.log("姓名：" + student.name);

console.log("年齡：" + student.age);

console.log("科系：" + student.department);
JSON 10test.js

JSON + object + array 綜合練習

JSON JSON.parse for if

用 JSON.parse 解析 JSON 字串，再用 for + if 判斷每位學生是否及格。

let jsonText = `
        
[

    {"name":"小明","score":85},
    
    {"name":"小華","score":58},
    
    {"name":"小美","score":92}
    
]

`;



let students = JSON.parse(jsonText);



for (let i = 0; i < students.length; i++) {

    if (students[i].score >= 60) {
    
        console.log(students[i].name + " 及格");
        
    } else {
    
        console.log(students[i].name + " 不及格");
        
    }
    
}
